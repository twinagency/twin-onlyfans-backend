COUNT = 0
CLONE = "11111111-1111-4111-8111-111111111111"
BINDING = "binding-worker"
BOT = "7123456789"
CHAT = "9001"
BC = "bc-worker"
ACCOUNT = "910001"
WORKER = "worker-one"


def check(value, label):
    global COUNT
    assert value, label
    COUNT += 1


def stamp(offset: int = 0) -> str:
    return (datetime.now(timezone.utc) + timedelta(seconds=offset)).isoformat(
        timespec="milliseconds"
    ).replace("+00:00", "Z")


def message_event(*, event_id: str, update_id: str, message_id: str,
                  text: str, origin: str, when: str) -> dict:
    return {
        "schemaVersion": 1, "eventId": event_id, "channel": "telegram",
        "bindingId": BINDING, "cloneId": CLONE, "botId": BOT,
        "mode": "business", "businessConnectionId": BC,
        "chatId": CHAT, "userId": CHAT, "updateId": update_id,
        "messageId": message_id, "occurredAt": when, "receivedAt": when,
        "origin": origin, "eventKind": "message", "text": text,
        "attachments": [], "sourceRevision": f"message:{message_id}:1",
    }


class FakeGateway:
    def __init__(self):
        self.events: list[dict] = []
        self.actions: list[dict] = []
        self.next_message = 8000

    def append(self, event_kind: str, payload: dict, source_id: str) -> None:
        self.events.append({
            "sourceEventId": source_id,
            "sourceSequence": len(self.events) + 1,
            "eventKind": event_kind,
            "payload": payload,
        })

    async def read_projection_events(self, *, after: int, limit: int) -> list[dict]:
        return [e for e in self.events if e["sourceSequence"] > after][:limit]

    async def submit_action(self, action: dict) -> dict:
        self.actions.append(action)
        self.next_message += 1
        delivered = stamp()
        external_id = str(self.next_message)
        receipt = message_event(
            event_id=f"receipt-{external_id}", update_id=f"9{external_id}",
            message_id=external_id, text=action["text"],
            origin="own_account", when=delivered,
        )
        self.append("telegram_receipt_v1", {
            "bindingId": BINDING, "cloneId": CLONE, "accountId": ACCOUNT,
            "event": receipt, "projectedAt": delivered,
        }, f"source-receipt-{external_id}")
        return {"state": "confirmed", "telegramMessageId": external_id,
                "deliveredAt": delivered}


async def create_schema(path: Path) -> None:
    from db.engine import engine, get_session
    from db.models import Base, Employee
    async with engine.begin() as connection:
        await connection.run_sync(Base.metadata.create_all)
    async with get_session() as session:
        session.add(Employee(id=1, display_name="Automation", is_active=True))


async def exercise(path: Path) -> None:
    from db.engine import engine, get_session
    from db.models import Fan, Message
    from sqlalchemy import select
    from telegram_runtime.worker import TelegramWorker, WorkerConfig

    gateway = FakeGateway()
    gateway.append("telegram_config_v1", {
        "bindingId": BINDING, "cloneId": CLONE, "accountId": ACCOUNT,
        "configVersion": "cfg-1", "creatorName": "Synthetic creator",
        "persona": "Warm, concise and curious.", "personaFacts": {},
        "language": "en", "timezone": None,
        "location": "Brisbane", "projectedAt": stamp(-3),
    }, "source-config-1")
    first_at = stamp(-60)
    gateway.append("telegram_inbound_v1", {
        "event": message_event(event_id="event-in-1", update_id="101",
                               message_id="501", text="How was your morning?",
                               origin="fan", when=first_at),
        "conversationKey": "conversation-worker", "inputRevision": 1,
        "ownershipEpoch": 4, "configVersion": "cfg-1",
        "projectedAt": stamp(-1),
    }, "source-in-1")

    model_calls: list[dict] = []
    answers = iter(("Glad you stopped by. How are you?", "Tell me more about that."))

    async def fake_chat(**kwargs):
        model_calls.append(kwargs)
        if kwargs.get("response_format"):
            return SimpleNamespace(content="{}", parsed={})
        return SimpleNamespace(content=next(answers), parsed=None)

    worker = TelegramWorker(
        database_path=str(path),
        config=WorkerConfig(
            worker_id=WORKER, account_id=ACCOUNT, binding_id=BINDING,
            clone_id=CLONE, model_id="openrouter/deepseek-v4-flash",
            credential_env="SYNTHETIC_OPENROUTER_KEY",
            provider_allowlist=frozenset({"openrouter"}),
            config_version="cfg-1",
        ),
        gateway=gateway, model_chat=fake_chat,
    )
    try:
        first = await worker.run_once()
        check(first["status"] == "completed", "first durable turn completes")
        check(first["actions"] == 2 and len(gateway.actions) == 2,
              f"first turn submits exactly the two native bubbles: {first!r} "
              f"actions={gateway.actions!r} model_calls="
              f"{[(sorted(c), c.get('response_format')) for c in model_calls]!r}")
        check([a["text"] for a in gateway.actions] ==
              ["Glad you stopped by", "How are you?"],
              "Telegram worker preserves Fastt bubbles without deliberate typos")
        check(gateway.actions[0]["expectedEpoch"] == 4,
              "action carries projected ownership epoch")
        check(gateway.actions[0]["expectedInputRevision"] == 1,
              "action carries projected input revision")
        check(gateway.actions[0]["configVersion"] == "cfg-1",
              "action carries projected config version")
        async with get_session() as session:
            rows = (await session.execute(
                select(Message).where(Message.account_id == ACCOUNT)
                .order_by(Message.created_at, Message.message_id)
            )).scalars().all()
            check(len(rows) == 3, "first inbound and two confirmed bubbles stored once")
            fan = await session.get(Fan, (ACCOUNT, rows[0].fan_id))
            fan.automation_paused_until = datetime.now(timezone.utc).replace(tzinfo=None) - timedelta(seconds=1)

        second_at = stamp(2)
        gateway.append("telegram_inbound_v1", {
            "event": message_event(event_id="event-in-2", update_id="102",
                                   message_id="502", text="I went for a walk.",
                                   origin="fan", when=second_at),
            "conversationKey": "conversation-worker", "inputRevision": 2,
            "ownershipEpoch": 4, "configVersion": "cfg-1",
            "projectedAt": stamp(3),
        }, "source-in-2")
        second = await worker.run_once()
        check(second["status"] == "completed", "second durable turn completes")
        check(len(gateway.actions) == 3, "second turn submits one additional action")
        async with get_session() as session:
            rows = (await session.execute(
                select(Message).where(Message.account_id == ACCOUNT)
            )).scalars().all()
            check(len(rows) == 5, "receipt replay creates no duplicate history row")
        prompts = str([call.get("messages") for call in model_calls])
        check("Glad you stopped by" in prompts,
              "second native prompt sees confirmed first reply")
        check("I went for a walk" in prompts,
              "second native prompt sees the fresh inbound")
        check(all(a["conversationKey"] == "conversation-worker" for a in gateway.actions),
              "all actions remain in the selected conversation")
        check(worker.store.get_cursor(WORKER) == len(gateway.events) - 1,
              "worker cursor stops before its newly emitted receipt")
        worker.recover_interrupted()
        check(worker.store.conn.execute(
            "select count(*) from telegram_runtime_turns where state='completed'"
        ).fetchone()[0] == 2, "completed turns remain terminal after recovery")
    finally:
        worker.close()
        await engine.dispose()


def main():
    import dotenv
    dotenv.load_dotenv = lambda *args, **kwargs: False
    logging.disable(logging.CRITICAL)
    # Keep warnings visible: a swallowed native boundary error must be diagnosable
    # in this acceptance process without exposing provider or fan data.
    with tempfile.TemporaryDirectory(prefix="telegram-runtime-worker-") as tmp:
        path = Path(tmp) / "native.db"
        with patch.dict(os.environ, {
            "TWIN_FASTT_CONFIG_ISOLATED": "1",
            "DATABASE_URL": "sqlite+aiosqlite:///" + path.as_posix(),
            "SYNTHETIC_OPENROUTER_KEY": "fixture-only",
            "PYTHON_DOTENV_DISABLED": "1",
        }, clear=True):
            async def guarded():
                await create_schema(path)
                def deny(*args, **kwargs):
                    raise AssertionError("network forbidden in worker fixture")
                with patch.object(socket.socket, "connect", deny), \
                     patch.object(socket.socket, "connect_ex", deny), \
                     patch.object(socket, "create_connection", deny):
                    await exercise(path)
            asyncio.run(guarded())
    print(f"{COUNT} checks; all cases passed (telegram runtime worker)")


if __name__ == "__main__":
    main()
