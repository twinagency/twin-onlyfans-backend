def rule(field, op, value, **extra):
    return {"match": "all", "rules": [{"field": field, "op": op, "value": value, **extra}]}


def save(store, name="Synthetic audience", rules=None, **extra):
    return save_audience(store, "tenant-a", "acc-a", actor=ACTOR, action="save", id=None,
        expected_revision=None, name=name, rules=rules or rule("purchase_count", "==", 0, days=30), **extra)["audience"]


def current(database):
    with Session(database[1]) as s, s.begin():
        runtime = s.get(OfRuntime, "acc-a")
        runtime.last_sync_at, runtime.recovery_needed = utcnow(), False

    assert (result["matched"], result["unknown"]) == (0, 1)

def test_scoped_lists_edit_conflicts_clone_retry_and_deleted_definition(database):
    store = setup(database)
    first = save(store)
    assert save(store)["id"] == first["id"]
    assert list_audiences(store, "tenant-a", "acc-a")["audiences"] == [first]
    newer = save_audience(store, "tenant-a", "acc-a", actor=ACTOR, action="save", id=first["id"],
        expected_revision=first["revision"], name="Edited audience", rules=rule("recent_spend", ">=", 1000, days=30))["audience"]
    with pytest.raises(Refused):
        save_audience(store, "tenant-a", "acc-a", actor=ACTOR, action="save", id=first["id"],
            expected_revision=first["revision"], name="Competing edit", rules=rule("status", "==", "active"))
    with pytest.raises(Refused):
        list_audiences(store, "another-tenant", "acc-a")
    token = current_scope.set(AgencyScope(("creator-acc-a",), "agency-user:00000000-0000-0000-0000-000000000001:00000000-0000-0000-0000-000000000002", False))
    try:
        assert list_audiences(store, "tenant-a", "acc-a")["audiences"][0]["id"] == first["id"]
        with pytest.raises(ValueError):
            save(store)
    finally:
        current_scope.reset(token)
    save_audience(store, "tenant-a", "acc-a", actor=ACTOR, action="remove", id=newer["id"], expected_revision=newer["revision"])
    assert list_audiences(store, "tenant-a", "acc-a")["audiences"] == []

def vault_seed(database):
    store = setup(database)
    now = utcnow()
    with Session(database[1]) as s, s.begin():
        s.add_all([VaultItem(account_id="acc-a",media_id=11,kind="photo",created_at=now), VaultItem(account_id="acc-a",media_id=12,kind="video",created_at=now)])
        s.add(Message(account_id="acc-a",fan_id=101,message_id=9001,direction="out",body="Synthetic bundle",media_ids="[11,12]",media_count=2,price_cents=1000,created_at=now-timedelta(minutes=2)))
        s.flush()
        s.add_all([VaultSend(account_id="acc-a",fan_id=101,message_id=9001,media_id=11,price_cents=1000,sent_at=now),
            VaultSend(account_id="acc-a",fan_id=101,message_id=9001,media_id=11,price_cents=1000,sent_at=now),
            MessageMedia(account_id="acc-a",fan_id=101,message_id=9001,media_id=11)])
    return store


def test_vault_sent_history_deduplicates_and_incomplete_negatives_remain_unknown(database):
    store = vault_seed(database)
    read = read_vault_intelligence(store,"tenant-a","acc-a",fan_id="101")
    assert len(read["items"]) == 2
    assert all(x["sentCount"] == 1 and x["ownership"] == "unknown" for x in read["items"])
    assert all(x["lastOfferedUnitPriceCents"] == 1000 for x in read["items"])
    assert read_vault_intelligence(store,"tenant-a","acc-a",fan_id="101",state="not_purchased")["items"] == []
    assert [x["mediaId"] for x in read_vault_intelligence(store,"tenant-a","acc-a",fan_id="101",kind="video")["items"]] == ["12"]
    with pytest.raises(Refused):
        read_vault_intelligence(store,"tenant-b","acc-a",fan_id="101")


def test_bundle_purchase_and_refund_are_attributed_once_per_unit(database):
    store = vault_seed(database)
    with Session(database[1]) as s, s.begin():
        item = CatalogItem(account_id="acc-a",label="Synthetic two-media bundle",kind="photo",media_ids="[11,12]",preview_media_ids="[]",price_cents=1000)
        s.add(item);s.flush()
        for _ in range(1):
            s.add(ContentOffer(account_id="acc-a",fan_id=101,item_id=item.id,mode="ppv",price_cents=1000,offer_message_id=9001,status="open"))
    paid = parsed(transaction(messageId=9001))
    assert record(store,"acc-a",paid) == "applied"
    with Session(database[1]) as s, s.begin():
        original = s.scalar(select(ContentOffer))
        s.add(ContentOffer(account_id="acc-a",fan_id=101,item_id=original.item_id,mode="ppv",price_cents=1000,offer_message_id=9001,status="open"))
    report = read_performance(store,"tenant-a","acc-a")
    assert len(report["items"]) == 1
    row = report["items"][0]
    assert (row["sentCount"],row["purchases"],row["clearedGrossCents"]) == (1,1,1000)
    history = read_vault_intelligence(store,"tenant-a","acc-a",fan_id="101")
    assert all(x["purchaseCount"] == 1 and x["lastPurchasedUnitPriceCents"] == 1000 for x in history["items"])
    # The reporting reader tolerates duplicate old offers; ledger admission
    # still refuses ambiguous ownership, so retain its existing contract.
    with Session(database[1]) as s, s.begin():
        duplicate = s.scalars(select(ContentOffer).order_by(ContentOffer.id.desc())).first()
        s.delete(duplicate)
    assert record(store,"acc-a",{**paid,"status":"refunded"}) == "applied"
    row = read_performance(store,"tenant-a","acc-a")["items"][0]
    assert (row["purchases"],row["refunds"],row["clearedGrossCents"]) == (0,1,0)


def test_large_preview_refuses_truncation_and_respects_contact_suppression(database):
    store = setup(database)
    current(database)
    now = utcnow()
    with Session(database[1]) as s, s.begin():
        s.add_all([Fan(account_id="acc-a",fan_id=i,is_muted=i == 1301) for i in range(102,1302)])
        s.flush()
        s.add_all([OfConversationState(account_id="acc-a",fan_id=i,mode="automated",takeover_state="none",
            updated_at=now) for i in range(102,1302)])
    saved = save(store)
    selection = {"include":[{"id":saved["id"],"revision":saved["revision"]}],"exclude":[]}
    statements = []
    def count(conn,cursor,statement,params,ctx,many):
        statements.append(statement)
    event.listen(store.engine,"before_cursor_execute",count)
    try:
        result = preview(store,"tenant-a","acc-a",selection=selection)
    finally:
        event.remove(store.engine,"before_cursor_execute",count)
    assert (result["total"],result["eligible"],result["suppressed"]) == (1201,1200,1)
    assert result["snapshotHash"] is None and len(result["sample"]) == 25
    assert len(statements) < 25, "Audience queries must not become per-fan reads"
    with pytest.raises((ValueError,Refused)):
        create(CampaignService(store),audience={"kind":"segment","selection":selection,"snapshotHash":"a"*64})

def test_populated_upgrade_preserves_rows_and_requires_compatible_rollback(database):
    store=setup(database)
    saved=save(store)
    from test_onlyfans_offer_import_upgrade_pg import historical_schema
    from db.schema import LEDGER_HISTORY,check_schema,upgrade,SchemaMismatch,_config
    from alembic import command
    with database[1].begin() as c:
        historical=historical_schema(c,LEDGER_HISTORY)
        before={t.name:[dict(r) for r in c.execute(t.select()).mappings()] for t in historical.sorted_tables if t.name != "alembic_version"}
        upgrade(c); upgrade(c); check_schema(c)
        for name,rows in before.items():
            assert [dict(r) for r in c.execute(historical.tables[name].select()).mappings()] == rows,name
        with pytest.raises(SchemaMismatch,match="not supported"):
            check_schema(c,expected=LEDGER_HISTORY)
        with pytest.raises(RuntimeError,match="compatible forward release"):
            command.downgrade(_config(c),LEDGER_HISTORY)
        check_schema(c)
    assert list_audiences(store,"tenant-a","acc-a")["audiences"][0]["id"] == saved["id"]
